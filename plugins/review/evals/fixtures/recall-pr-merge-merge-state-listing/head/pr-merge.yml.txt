      # Candidates: open, non-draft PRs into main that the lanes App
      # (merge.agent-identity on main) opened, plus Dependabot's unless
      # merge.dependabot.mode is off. This is only a shortlist; the gate checks
      # the author, the closing issue and verifier verdict or the Dependabot
      # tier, and every other rule again. A candidate the gate would refuse on
      # its listed labels or body alone (the `do-not-merge` label, the
      # human-gated role label from .work-item-tracker.json, `needs-human` by
      # default, or a hold in the body) is set aside here before it costs
      # lane-token calls (babysit_merge_prefilter.py); the summary still
      # reports it. Read with the job's own read-only token.
      - name: List the candidate PRs
        id: candidates
        if: steps.lane-token.outputs.token != ''
        env:
          GH_TOKEN: ${{ github.token }}
          REPO: ${{ github.repository }}
          DEPENDABOT: ${{ steps.mode.outputs.dependabot }}
        run: |
          set -euo pipefail
          identity=$(yq '.merge."agent-identity" // ""' .base/docs/conventions/pr-pipeline.yaml)
          slug=${identity%\[bot\]}
          if ! [[ $slug =~ ^[a-z0-9][a-z0-9-]*$ ]]; then
            echo "::error::merge.agent-identity is not an App bot login: $identity"
            exit 1
          fi
          authors=("app/$slug")
          # The gate resolves the same label through babysit_postmerge._role_label.
          human_gated=needs-human
          if [ -f .base/.work-item-tracker.json ]; then
            human_gated=$(jq -er '[objects | .config | objects | .role_labels | objects
              | .["human-gated"] | strings | gsub("^\\s+|\\s+$"; "")][0] // ""
              | if . == "" then "needs-human" else . end' \
              .base/.work-item-tracker.json) || {
              echo "::error::cannot read the human-gated role label from the base .work-item-tracker.json"
              exit 1
            }
          fi
          if [ "$DEPENDABOT" != off ]; then
            authors+=(app/dependabot)
          fi
          for author in "${authors[@]}"; do
            gh pr list --repo "$REPO" --base main --state open --limit 500 \
              --author "$author" --json number,headRefOid,isDraft,updatedAt,labels,body,createdAt,mergeStateStatus
          done |
            jq -cs '[add[] | select(.isDraft | not)
              | {number, head: .headRefOid, updatedAt, labels: [.labels[].name], body, createdAt, mergeStateStatus}]
              | unique_by(.number)' |
            python3 .base/plugins/source-control/skills/babysit-prs/scripts/babysit_merge_prefilter.py \
              --human-gated="$human_gated" >"$RUNNER_TEMP/prefilter.json"
          jq -c .candidates "$RUNNER_TEMP/prefilter.json" >"$RUNNER_TEMP/candidates.json"
          jq -c .held "$RUNNER_TEMP/prefilter.json" >"$RUNNER_TEMP/held.json"
