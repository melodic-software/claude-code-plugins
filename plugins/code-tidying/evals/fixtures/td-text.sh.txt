#!/usr/bin/env bash
# Shared text helpers. Sourced by scripts in this repo and, by function name,
# by the deploy-tools and infra-scripts repositories.

fmtDate() {
  date -u -d "@$1" +%Y-%m-%d
}

trim_space() {
  local s="$1"
  s="${s#"${s%%[![:space:]]*}"}"
  s="${s%"${s##*[![:space:]]}"}"
  printf '%s' "$s"
}
