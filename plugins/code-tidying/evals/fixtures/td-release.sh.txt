#!/usr/bin/env bash
# Validate VERSION and print the tag a release would create.
set -euo pipefail

old_registry="registry.example.invalid"

# Get the version
version="$(cat VERSION)"
if [[ -n "$version" ]]; then
  if [[ "$version" =~ ^[0-9]+\.[0-9]+\.[0-9]+$ ]]; then
    echo "would tag v$version"
  else
    echo "bad version: $version" >&2
    exit 1
  fi
fi
