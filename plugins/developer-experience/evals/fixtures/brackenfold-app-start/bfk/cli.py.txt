"""Command-line entry point: `bfk report`, `bfk release`, `bfk app start`."""

import argparse
import sys

from bfk import app, release, store


def main(argv=None):
    parser = argparse.ArgumentParser(prog="bfk", description="brackenfold team tooling")
    sub = parser.add_subparsers(dest="command", required=True)
    sub.add_parser("report", help="print a summary of data/accounts.jsonl")
    rel = sub.add_parser("release", help="tag a release (asks for confirmation)")
    rel.add_argument("version", help="version to tag, for example 0.5.0")
    app_parser = sub.add_parser("app", help="manage the app server")
    app_sub = app_parser.add_subparsers(dest="app_command", required=True)
    app_sub.add_parser("start", help="start the app server in the background")
    args = parser.parse_args(argv)

    if args.command == "report":
        accounts = store.load_accounts()
        domains = sorted({a["email"].split("@", 1)[1] for a in accounts})
        print(f"{len(accounts)} accounts across {len(domains)} domains: {', '.join(domains)}")
        return 0
    if args.command == "release":
        return release.run(args.version)
    if args.command == "app":
        return app.start()
    return 2


if __name__ == "__main__":
    sys.exit(main())
